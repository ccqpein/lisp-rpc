run the generator with the spec in this folder

For debugging:

#+begin_src shell
  ../../../target/debug/lisp-rpc-rust-generator \
          --input-file spec.lisprpc\
          --templates-path ../../../generators/lisp-rpc-rust-generator/templates/ \
          --output-path .
#+end_src

#+RESULTS:

#+begin_src shell
  ../../../target/debug/lisp-rpc-rust-generator \
          --input-file spec.lisprpc\
          --output-path ~/Desktop/
